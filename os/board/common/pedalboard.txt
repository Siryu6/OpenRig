# Pedalboard settings, read at boot. This file is on the small FAT partition
# of the card: edit it from any computer, then restart the pedalboard.

# name as listed in /proc/asound/cards, e.g. AudioBox or pisound

# Sample rate (Hz) and buffer (period frames x periods). 128 x 2 at 48 kHz
# adds about 5 ms each way; use 256 if you hear clicks.

period=128
periods=2

display_rotation=0
